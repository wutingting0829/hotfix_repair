#include "lua.h"
#include "lauxlib.h"
#include <stdint.h>
#include <limits.h>  // For INT_MIN

typedef int32_t SBits;
typedef uint32_t UBits;

// Fix for potential buffer overflow in bit_tohex function
static int bit_tohex(lua_State *L)
{
  UBits b = barg(L, 1);
  SBits n = lua_isnone(L, 2) ? 8 : (SBits)barg(L, 2);
  const char *hexdigits = "0123456789abcdef";
  char buf[8];
  int i;

  // Ensure n is within valid range to prevent buffer overflow
  if (n < 0) {
    if (n == INT_MIN) n = 8;  // Handle INT_MIN case to avoid overflow
    else n = -n;
    hexdigits = "0123456789ABCDEF";
  }
  if (n > 8) n = 8;

  for (i = (int)n; --i >= 0; ) {
    buf[i] = hexdigits[b & 15];
    b >>= 4;
  }
  lua_pushlstring(L, buf, (size_t)n);
  return 1;
}
